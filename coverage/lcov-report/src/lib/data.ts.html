function p(
  id: number,
  name: string,
  category: string,
  price: number,
  mrp: number,
  rating: number,
  reviews: number,
  emoji: string,
  description: string,
  tags: Product["tags"],
): Product {
  return {
    id: `p${id}`,
    slug: name
      .toLowerCase()
      .replace(/[^a-z0-9]+/g, "-")
      .replace(/(^-|-$)/g, ""),
    name,
    category,
    price,
    mrp,
    rating,
    reviews,
    stock: 8 + ((id * 7) % 40),
    emoji,
    description,
    tags,
  };
}
